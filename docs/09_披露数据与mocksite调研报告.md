def _generate_spot_clearance_data() -> list[dict]:
    """生成现货出清电量数据（24小时）"""
    data = []
    base_energy = 80000  # 基础电量
    for hour in range(24):
        variance = random.uniform(-0.3, 0.3)  # ±30% 波动
        energy = base_energy * (1 + variance)
        units = int(250 + (energy - 50000) / 1000)  # 与电量正相关
        data.append({
            "时刻": f"{hour}:00",
            "电量(MWh)": round(energy, 2),
            "开机台数(台)": units
        })
    return data

def _generate_market_electricity_data() -> dict:
    """生成市场用电信息（24小时）"""
    base_demand = 100000
    demand = []
    actual = []
    for hour in range(24):
        # 日间系数（10-12点和17-19点是峰值）
        if hour in (10, 11, 17, 18, 19):
            coeff = 1.15
        elif hour in (2, 3, 4):
            coeff = 0.85
        else:
            coeff = 1.0
        
        hour_demand = base_demand * coeff * random.uniform(0.95, 1.05)
        hour_actual = hour_demand * random.uniform(0.87, 0.93)  # 需求-实际 gap
        
        demand.append(round(hour_demand, 4))
        actual.append(round(hour_actual, 4))
    
    return {
        "需求": demand,
        "实际": actual
    }

**实现注意事项**:
- 不需要全部 2838 个节点，可简化为 50 个代表节点
- 时间戳采用 15 分钟间隔（00:00, 00:15, ... 23:45）

- 相邻时间点电价波动通常不超过 ±100 元/MWh

**简化版实现**:
```python
def _generate_node_prices_simplified(num_nodes: int = 50) -> list[dict]:
    """简化版节点电价生成（50个代表节点）"""
    nodes = [
        "其他500kV水城变500kV#1M",
        "其他一总降站220_KV#1",
        ... # 共50个节点
    ]
    
    data = []
    for node in nodes[:num_nodes]:
        prices = []
        base = random.uniform(350, 450)
        for i in range(96):  # 96个15分钟
            # 00-03点有异常值
            if i < 12:  # 前3小时
                price = -50 if random.random() < 0.1 else base
            else:
                # 正常时段，加入日间/夜间系数
                hour = i // 4
                if hour in (11, 12, 13, 18, 19, 20):
                    coeff = 1.2  # 峰值
                else:
                    coeff = 1.0
                price = base * coeff + random.uniform(-50, 50)
            
            prices.append(round(price, 2))
        
        data.append({
            "节点名称": node,
            "电价(元/MWh)": prices
        })
    
    return data
```
